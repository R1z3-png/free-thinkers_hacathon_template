cmake_minimum_required(VERSION 3.22.1)
project(llamacpp_jni LANGUAGES C CXX)
include(FetchContent)
# KleidiAI has deeply nested filenames; use the OS temp directory to keep its
# generated object paths below Windows' legacy 260-character limit.
if(DEFINED ENV{TEMP})
  file(TO_CMAKE_PATH "$ENV{TEMP}" _fetch_root)
elseif(DEFINED ENV{TMPDIR})
  file(TO_CMAKE_PATH "$ENV{TMPDIR}" _fetch_root)
else()
  set(_fetch_root "${CMAKE_CURRENT_LIST_DIR}/../../../.cmake-fetch")
endif()
set(FETCHCONTENT_BASE_DIR "${_fetch_root}/free-thinkers-fetch-${CMAKE_PROJECT_NAME}" CACHE PATH "" FORCE)
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# Match the optimized x86_64 kernels used by whisper.cpp on the emulator.
# Keep these target features scoped to x86_64; the arm64 build stays portable.
if(ANDROID_ABI STREQUAL "x86_64")
  set(GGML_SSE42 ON CACHE BOOL "" FORCE)
  set(GGML_AVX ON CACHE BOOL "" FORCE)
  set(GGML_AVX2 ON CACHE BOOL "" FORCE)
  set(GGML_BMI2 ON CACHE BOOL "" FORCE)
  set(GGML_FMA ON CACHE BOOL "" FORCE)
  set(GGML_F16C ON CACHE BOOL "" FORCE)
endif()
if(ANDROID_ABI STREQUAL "arm64-v8a")
  # Keep the baseline portable while enabling Arm's runtime-selected kernels
  # for supported operations and CPU features (with a generic fallback).
  set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
endif()
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
if(DEFINED ENV{DICTOPHONE_LLAMA_CPP_DIR})
  file(TO_CMAKE_PATH "$ENV{DICTOPHONE_LLAMA_CPP_DIR}" _llama_source)
  FetchContent_Declare(llama SOURCE_DIR "${_llama_source}")
else()
  FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b12189 GIT_SHALLOW TRUE GIT_SUBMODULES_RECURSE TRUE)
endif()
FetchContent_MakeAvailable(llama)
add_library(llamacpp_jni SHARED llama_jni.cpp)
target_link_libraries(llamacpp_jni PRIVATE llama log android)
target_compile_options(llamacpp_jni PRIVATE -O3 -fvisibility=hidden)
