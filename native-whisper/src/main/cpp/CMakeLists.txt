cmake_minimum_required(VERSION 3.22.1)
project(whispercpp_jni LANGUAGES C CXX)
include(FetchContent)
# KleidiAI has deeply nested filenames; use the OS temp directory to keep its
# generated object paths below Windows' legacy 260-character limit.
if(DEFINED ENV{TEMP})
  file(TO_CMAKE_PATH "$ENV{TEMP}" _fetch_root)
elseif(DEFINED ENV{TMPDIR})
  file(TO_CMAKE_PATH "$ENV{TMPDIR}" _fetch_root)
else()
  set(_fetch_root "${CMAKE_CURRENT_LIST_DIR}/../../../.cmake-fetch")
endif()
set(FETCHCONTENT_BASE_DIR "${_fetch_root}/free-thinkers-fetch-${CMAKE_PROJECT_NAME}" CACHE PATH "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(WHISPER_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(WHISPER_ALL_WARNINGS_3RD_PARTY OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# The x86_64 Android emulator exposes AVX2, FMA, F16C, and BMI2. The generic
# cross-compiled CPU backend otherwise disables these kernels and is very slow.
# Keep the optimized instruction set scoped to x86_64; arm64 builds remain portable.
if(ANDROID_ABI STREQUAL "x86_64")
  set(GGML_SSE42 ON CACHE BOOL "" FORCE)
  set(GGML_AVX ON CACHE BOOL "" FORCE)
  set(GGML_AVX2 ON CACHE BOOL "" FORCE)
  set(GGML_BMI2 ON CACHE BOOL "" FORCE)
  set(GGML_FMA ON CACHE BOOL "" FORCE)
  set(GGML_F16C ON CACHE BOOL "" FORCE)
endif()
if(ANDROID_ABI STREQUAL "arm64-v8a")
  # Keep the baseline portable while enabling Arm's runtime-selected kernels
  # for supported operations and CPU features (with a generic fallback).
  set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
endif()
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
if(DEFINED ENV{DICTOPHONE_WHISPER_CPP_DIR})
  file(TO_CMAKE_PATH "$ENV{DICTOPHONE_WHISPER_CPP_DIR}" _whisper_source)
  FetchContent_Declare(whisper SOURCE_DIR "${_whisper_source}")
else()
  FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggerganov/whisper.cpp.git GIT_TAG v1.9.5 GIT_SHALLOW TRUE GIT_SUBMODULES_RECURSE TRUE)
endif()
FetchContent_MakeAvailable(whisper)
# whisper.cpp and llama.cpp are pinned independently and ship separate ggml builds.
# Give whisper's shared ggml libraries unique SONAMEs so both runtimes can coexist.
foreach(_target ggml ggml-base ggml-cpu)
  if(TARGET ${_target})
    set_target_properties(${_target} PROPERTIES OUTPUT_NAME "whisper_${_target}")
  endif()
endforeach()
add_library(whispercpp_jni SHARED whisper_jni.cpp)
target_link_libraries(whispercpp_jni PRIVATE whisper log android)
target_compile_options(whispercpp_jni PRIVATE -O3 -fvisibility=hidden)
